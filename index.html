// Define collections cleanly in code with strict TypeScript
import { buildConfig } from 'payload/config';
import { PostCollection } from './collections/Posts';

export default buildConfig({
  serverURL: 'http://localhost:3000',
  admin: {
    user: 'users',
    components: {
      graphics: { Logo: CustomBrandLogo },
    },
  },
  collections: [
    {
      slug: 'posts',
      access: {
        read: () => true, // Public read access
      },
      fields: [
        { name: 'title', type: 'text', required: true },
        { name: 'author', type: 'relationship', relationTo: 'users' },
        { name: 'content', type: 'richText' },
        { name: 'tags', type: 'array', fields: [{ name: 'tag', type: 'text' }] },
      ],
    },
  ],
  typescript: {
    outputFile: 'path/to/payload-types.ts',
  },
});
// Mount Payload inside your standard Express application
import express from 'express';
import payload from 'payload';

const app = express();

const start = async () => {
  // Initialize Payload with your configuration
  await payload.init({
    secret: process.env.PAYLOAD_SECRET,
    express: app,
    onInit: () => {
      payload.logger.info('🚀 Payload CMS running at http://localhost:3000');
    },
  });

  // Add custom Express routes, authentication, or middleware freely
  app.get('/api/health', (req, res) => {
    res.status(200).json({ status: 'online', database: 'connected' });
  });

  app.listen(3000);
};

start();

// 01. ARCHITECTURE

Payload is a headless CMS and application framework.

Unlike traditional monolithic CMS platforms or SaaS APIs, Payload gives you a full-stack engine that runs directly alongside your application code.

Total Freedom

Give you full control over your Express app

Payload doesn’t hide your server in an inaccessible cloud container. You initialize Payload directly within an Express instance, giving you the ability to attach your own routes, security layers, webhooks, and microservices in pure TypeScript.

  • Zero black-box abstractions or locked backends
  • Full access to Express req and res objects
  • Use any Node.js package or npm library directly
Database Agnostic

MongoDB, PostgreSQL, SQLite, or MySQL

Choose the database that fits your architecture. Seamlessly connect MongoDB or relational SQL engines without altering your collection field definitions.

// 02. COMPREHENSIVE TOOLKIT

Everything you need in a modern CMS.

Engineered to satisfy demanding production workloads without sacrificing developer delight.

GraphQL, REST & Local API

Payload automatically generates fully-typed GraphQL and REST endpoints based on your schema. Or use the Local API in your server without HTTP overhead.

Extensible Authentication

Complete authentication out of the box with JWT, email verification, password resets, API keys, and custom granular access control functions.

End-to-End TypeScript

Write your config in TypeScript and Payload automatically generates corresponding TypeScript interfaces. Never write duplicate types again.

Hooks & Middleware

Execute custom business logic at precise lifecycle moments: beforeChange, afterChange, beforeRead, afterDelete, and more.

Built-in File Uploads

Upload images, PDFs, and files directly. Automatic image resizing, crop focal points, MIME-type validation, and native S3/Cloud storage adapters.

White-Label Admin UI

Easily replace branding, inject custom React views, dashboard statistics widgets, and bespoke sidebar navigation to suit your client or team needs.
